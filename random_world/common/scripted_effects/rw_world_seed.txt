# ==============================================================================
# RANDOM WORLD - THE WORLD SEED (the one number you edit to get a new world)
# ------------------------------------------------------------------------------
# The world is generated BEFORE the country-selection lobby, and at that
# moment the game's random generator is intentionally deterministic (a fixed
# engine seed - the same trick vanilla uses for its weather seed 22345).
# Consequence: the number below fully decides WHICH world you get.
#
#     >>>  CHANGE THE NUMBER (any whole number from 0 to 999),  <<<
#     >>>  RESTART HOI4, START A NEW GAME - DIFFERENT WORLD     <<<
#
# The full game restart is REQUIRED: HOI4 parses mod files exactly once,
# during application launch. Edits made while the game is running are
# invisible to it - "Back to Menu -> New Game" is not enough.
#
# LAZY OPTION: double-click randomize_seed_and_play.bat in the mod root -
# it writes a random number here and starts the game for you, so every
# launch is a new world with zero manual editing.
#
# Same number = exactly the same world, every time. That gives you:
#   - reproducible worlds: write a number down to replay that world later;
#   - multiplayer safety: every player's game computes the identical world,
#     nothing to coordinate beyond running the same mod version.
#
# (This number only matters if you ENABLED the optional lobby-time trigger
# in common/bookmarks/the_gathering_storm.txt. In the default mode - the
# in-game "Randomize the World!" decision - the generator is already fresh
# every session and this number is irrelevant.)
# ==============================================================================

rw_set_world_seed = {
	set_variable = { global.rw_world_seed = 42 }
}
