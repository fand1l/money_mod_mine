# ==============================================================================
# RANDOM WORLD - 1936 scenario override
# ------------------------------------------------------------------------------
# Files in common/bookmarks are replaced BY FILENAME, so this file overrides
# the vanilla 1936 scenario ("the gathering storm") completely. Three changes:
#
#   1. NO featured countries. Vanilla lists GER/SOV/USA/... here - that is
#      what draws the big portrait cards on the Select Country screen.
#      We list none, so only the "Other countries" card remains and you go
#      straight to picking a country on the map.
#   2. default_country = "---" (= "other countries"). If your error.log ever
#      complains about it, replace "---" with any tag, e.g. NES.
#   3. The effect block below contains an OPTIONAL, DISABLED lobby-time
#      trigger for the reshuffle. The active method is the "Randomize the
#      World!" decision (Decisions panel, once per campaign, fresh world
#      every time). Uncommenting the block below instead reshuffles during
#      scenario setup, BEFORE the map lobby - handy for picking countries
#      on the finished map (multiplayer) - but at that moment the engine's
#      generator is deterministic: WHICH world appears is chosen by the
#      number in common/scripted_effects/rw_world_seed.txt, and the same
#      number always repeats the same world (change it + restart HOI4, or
#      use randomize_seed_and_play.bat).
#
# IMPORTANT: if your game version names this file differently, the override
# will not load. Open the GAME's folder common/bookmarks/ and make sure the
# mod's two files have EXACTLY the same names as the vanilla ones
# (1936 = "the_gathering_storm.txt", 1939 = "blitzkrieg.txt").
# ==============================================================================

bookmarks = {
	bookmark = {
		name = "GATHERING_STORM_NAME"
		desc = "GATHERING_STORM_DESC"
		date = 1936.1.1.12
		picture = "GFX_select_date_1936"
		default_country = "---"
		default = yes

		"---" = {
			history = "OTHER_GATHERING_STORM_DESC"
		}

		effect = {
			randomize_weather = 22345 # <- DO NOT REMOVE
			# OPTIONAL lobby-time reshuffle - DISABLED by default (see note 3
			# in the header). Remove the # from the five lines below to
			# generate the world before the lobby (world chosen by
			# rw_world_seed.txt; identical per number). While disabled, the
			# reshuffle is triggered by the in-game decision instead.
			# (random_country is required: scripted effects cannot be called
			# from this scope-less context.)
			#random_country = {
			#	if = {
			#		limit = { NOT = { has_global_flag = rw_world_randomized } }
			#		rw_randomize_world = yes
			#	}
			#}
		}
	}
}
