# ==============================================================================
# RANDOM WORLD - FALLBACK activation decision
# ------------------------------------------------------------------------------
# Normally you never see this: the world is randomized automatically by
# on_startup right after pressing Play (fresh random seed every launch).
# rw_randomize_world sets the global flag when it really runs, and that
# flag hides this decision and its whole category.
#
# This is the last-resort SAFETY NET: it can only be visible if the
# automatic trigger failed to execute (e.g. the mod's script files did not
# load). In that case you can still trigger the reshuffle manually:
#
#   - costs 0 political power, available from day 1;
#   - fire_only_once + the global flag make it strictly one-shot per
#     campaign (save/load can never re-run it);
#   - ai_will_do = 0 means AI countries never press it.
# ==============================================================================

RW_random_world_category = {

	RW_randomize_world_now = {
		icon = generic_civil_support

		allowed = { always = yes }

		visible = {
			NOT = { has_global_flag = rw_world_randomized }
		}

		cost = 0
		fire_only_once = yes

		ai_will_do = {
			factor = 0
		}

		complete_effect = {
			rw_randomize_world = yes
		}
	}
}
