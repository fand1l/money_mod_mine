# ==============================================================================
# RANDOM WORLD - FALLBACK activation decision
# ------------------------------------------------------------------------------
# Normally you never see this: the world is randomized automatically while
# the 1936 scenario loads (common/bookmarks/the_gathering_storm.txt sets the
# global flag, which hides this decision and its whole category).
#
# It exists as a SAFETY NET: if the bookmark override failed to load on your
# game version (wrong filename - see the note in that file), the flag is
# never set, the "Random World" category appears in the Decisions panel, and
# you can still trigger the reshuffle manually:
#
#   - costs 0 political power, available from day 1;
#   - fire_only_once + the global flag make it strictly one-shot per
#     campaign (save/load can never re-run it);
#   - ai_will_do = 0 means AI countries never press it.
# ==============================================================================

RW_random_world_category = {

	RW_randomize_world_now = {
		icon = generic_civil_support

		allowed = { always = yes }

		visible = {
			NOT = { has_global_flag = rw_world_randomized }
		}

		cost = 0
		fire_only_once = yes

		ai_will_do = {
			factor = 0
		}

		complete_effect = {
			set_global_flag = rw_world_randomized
			rw_randomize_world = yes
		}
	}
}
