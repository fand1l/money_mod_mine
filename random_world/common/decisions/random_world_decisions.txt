# ==============================================================================
# RANDOM WORLD - THE activation decision (the primary trigger)
# ------------------------------------------------------------------------------
# The world reshuffle is started MANUALLY from here: Decisions panel ->
# "Random World" category -> "Randomize the World!", by whatever country
# the human player controls. Because it fires in-session, the random
# generator is fresh every time - every campaign gets a different world,
# no seeds or restarts needed.
#
#   - costs 0 political power, available from day 1;
#   - fire_only_once + the global flag make it strictly one-shot per
#     campaign (save/load can never re-run it) - the flag also hides the
#     decision and its category once the world has been reshuffled;
#   - ai_will_do = 0 means AI countries never press it - only a human
#     decides when (and whether) the world gets reshuffled.
#
# An optional automatic lobby-time trigger (world generated before country
# selection, fixed per seed number) sits disabled in
# common/bookmarks/the_gathering_storm.txt.
# ==============================================================================

RW_random_world_category = {

	RW_randomize_world_now = {
		icon = generic_civil_support

		allowed = { always = yes }

		visible = {
			NOT = { has_global_flag = rw_world_randomized }
		}

		cost = 0
		fire_only_once = yes

		ai_will_do = {
			factor = 0
		}

		complete_effect = {
			rw_randomize_world = yes
		}
	}
}
