# ==============================================================================
# RANDOM WORLD - on_actions
# ------------------------------------------------------------------------------
# "on_actions" are hooks the game itself calls at fixed moments.
#
#   on_startup  - runs when a game session starts AND every time a save is
#                 loaded. It does three things:
#                   1. writes an [RW] line to game.log unconditionally - the
#                      definitive "is the mod loaded at all?" probe;
#                   2. enforces the faction ban (idempotent, no guard needed);
#                   3. ATTEMPT #2 (of 3) to run the world reshuffle: if the
#                      bookmark effect (attempt #1, before the lobby) was
#                      inert on this engine build, this fires it at session
#                      start - i.e. right after pressing Play. The flag set
#                      inside rw_randomize_world keeps it strictly one-shot
#                      per campaign, including across save loads.
#
#   on_weekly   - runs for EVERY country once each in-game week. We use it to
#                 re-apply the "no factions" rules, so countries that are
#                 created LATER (civil wars, released nations) can never
#                 create or join factions either.
#
# The file name starts with ZZ_ so it loads after vanilla files (purely
# cosmetic; on_actions from different files are merged, not replaced).
# ==============================================================================

on_actions = {

	on_startup = {
		effect = {
			log = "[RW] on_startup fired - mod files are loaded"
			# on_startup also runs with NO active scope, and scripted effects
			# cannot be invoked from a scope-less context (error.log says
			# "Invalid Scope ... provided: None"). Stepping into any existing
			# country first fixes that; the effects themselves operate on the
			# whole world.
			random_country = {
				rw_disable_factions_for_all = yes
				if = {
					limit = { NOT = { has_global_flag = rw_world_randomized } }
					rw_randomize_world = yes
				}
			}
		}
	}

	on_weekly = {
		effect = {
			# Scope here is the country whose week ticked.
			set_rule = {
				can_create_factions = no
				can_join_factions = no
			}
		}
	}
}
